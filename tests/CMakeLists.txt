cmake_minimum_required(VERSION 3.22)
project(ttc_native_tests C)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(lib "${CMAKE_CURRENT_SOURCE_DIR}/../Libraries")
add_executable(test_ttc test_ttc.c "${lib}/ttc_link.c" "${lib}/ttc_vote.c")
target_include_directories(test_ttc PRIVATE "${lib}")
if(NOT MSVC)
  target_link_libraries(test_ttc PRIVATE m)
endif()
add_executable(test_store test_store.c "${lib}/ttc_store.c" "${lib}/ttc_vote.c" "${lib}/ttc_link.c")
target_include_directories(test_store PRIVATE support "${lib}")
foreach(target test_ttc test_store)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
add_test(NAME ground_interop COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/interop.py" "$<TARGET_FILE:test_ttc>")
